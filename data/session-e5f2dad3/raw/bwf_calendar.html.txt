<script>
    Vue.component('date-picker-input', {
        template: '#date-picker-input',
        props: {
            value: {
                type: String,
                default: '',
            },
            label: {
                type: String,
                default: '',
            },
        },
        data() {
            return {
                datePickerModel: false,
                localValue: this.value, // Copy the prop to a local data property
                dateFormatted: this.formatDate(this.value),
                dateError: null, // For storing validation errors
            };
        },
        watch: {
            value(newValue) {
                this.localValue = newValue; // Watch for changes in the prop
                this.dateFormatted = this.formatDate(this.localValue);
            },
            localValue(newValue) {
                this.$emit('input', newValue);
                this.datePickerModel = false;
                this.dateFormatted = this.formatDate(newValue); // Update formatted value
            }
        },
        methods: {
            clearClick(e) {
                this.localValue = null;
                this.dateFormatted = ''; // Clear the formatted date
            },
            formatDate(date) {
                if (!date || date.length === 0) return '';

                const [year, month, day] = date.split('-');
                return `${day}/${month}/${year}`;
            },
            validateDate(date) {
                if (!date){
                    //allow blank & clearing
                    return true;
                }
                const [day, month, year] = date.split('/').map(Number);

                const parsedDate = new Date(year, month - 1, day); // month is 0-indexed

                if (
                    parsedDate.getFullYear() === year &&
                    parsedDate.getMonth() === month - 1 &&  // month is 0-indexed
                    parsedDate.getDate() === day
                ) {
                    return true;
                }
                return false;
            },
            validateAndUpdate() {
               if (this.validateDate(this.dateFormatted)) {
                   if (this.dateFormatted){
                        const [day, month, year] = this.dateFormatted.split('/');
                        this.localValue = `${year}-${month}-${day}`; // Convert to ISO format (yyyy-mm-dd)
                   } else {
                       this.localValue = null;
                   }
                    this.dateError = null;
                } else {
                    this.dateError = 'Invalid date format. Use dd/mm/yyyy';
                }
            },
            onDatePickerInput(date) {
                this.dateFormatted = this.formatDate(date);
            }
        },
    });

<script>
    Vue.component('vue-menu-tabs', {
        template: '#vue-menu-tabs',
        props: {
            tabs: {
                type: Array,
                required: true // [{ value, label }]
            },
            value: {
                type: [String, Number],
                default: null
            },
            maxVisible: {
                type: Object,
                default: function() {
                    return {
                        sm: 2,
                        xs: 1
                    };
                }
            }
        },
        computed: {
            isOverflowTabActive() {
                return this.overflowTabs.findIndex(t => t.value === this.value) != -1;
            },
            limit: function() {
                const bp = this.$vuetify.breakpoint;
                if (bp.mdAndUp) return this.tabs.length; // show all
                if (bp.xs) return this.maxVisible.xs || this.maxVisible.sm || 2;
                if (bp.sm) return this.maxVisible.sm || 2;
                return this.tabs.length;
            },
            visibleTabs: function() {
                return this.tabs.slice(0, this.limit);
            },
            overflowTabs: function() {
                return this.tabs.slice(this.limit);
            },
            activeIndex: function() {
                const idx = this.visibleTabs.findIndex(t => t.value === this.value);
                return idx !== -1 ? idx : (this.overflowTabs.length ? this.visibleTabs.length : idx);
            }
        },
        methods: {
            select: function(val) {
                this.$emit('input', val);
                this.$emit('change', val);
            }
        }
    });

<script type="text/javascript">
    Vue.component('tournament-card', {
        template: '#tournament-card-template',
        props: {
            tmtItem: {
                type: Object,
                required: true
            },
            showEtihad: {
                type: Boolean,
                default: false
            },
            mirrored: {
                type: Boolean,
                default: false
            },
            etihadLeft: {
                type: Boolean,
                default: false
            },
        },
        data() {
            return {
                addToCalendarDialog : false,
            };
        },
        computed: {
            liveStatusClass(){
                return 'date-' + this.tmtItem.live_status;
            },
            showStatusLabel() {
                if (!this.tmtItem || !this.tmtItem.status){
                    return;
                }
                var status = this.tmtItem.status;
                var isSpecialStatus = status.code !== 'finished' && status.code !== 'normal' ;
                return isSpecialStatus &&  status.label !== null;
            },
            showLiveLabel() {
                return this.tmtItem.has_live_scores && ['cancelled', 'postponed'].indexOf(this.tmtItem.status.code) === -1;
            },
            showAddToCalendar(){
                return this.tmtItem.live_status !== 'post';
            },
            isAirBadminton(){
                // return this.tmtItem.category === "AirBadminton Team";
                return false;
            }
        },
        methods: {
            formatDateForGoogleCalendar(dateString, addDay, joinStr = '') {
                var date = new Date(dateString);
                if (addDay) {
                    date.setDate(date.getDate() + 1); // Add one day to the date
                }

                const year = date.getFullYear();
                const month = String(date.getMonth() + 1).padStart(2, '0'); // Months are 0-based, so we add 1
                const day = String(date.getDate()).padStart(2, '0');

                return `${year}${joinStr}${month}${joinStr}${day}`;
            },
            formatCalendarDescription(tmtItem){
                return `Follow BWF Tournament Live Scores: ${tmtItem.url}`;
            },
            addToGoogleCalendar(tmtItem) {
                const startDate = this.formatDateForGoogleCalendar(tmtItem.start_date);
                const endDate = this.formatDateForGoogleCalendar(tmtItem.end_date, true); // The day after the intended end date


                const baseUrl = 'https://www.google.com/calendar/render';
                const queryParams = new URLSearchParams({
                    action: 'TEMPLATE',
                    text: tmtItem.name,
                    dates: `${startDate}/${endDate}`,
                    details: this.formatCalendarDescription(tmtItem),
                    location: tmtItem.location,
                    sf: 'true',
                    output: 'xml'
                });
                const url = `${baseUrl}?${queryParams.toString()}`;

                window.open(url, '_blank');

                this.closeAddToCalendarDialog();
            },
            addToOutlook(tmtItem) {
                const startDate = this.formatDateForGoogleCalendar(tmtItem.start_date, false, '-');
                const endDate = this.formatDateForGoogleCalendar(tmtItem.end_date, true, '-'); // The day after the intended end date

                const baseUrl = 'https://outlook.live.com/calendar/action/compose';
                const queryParams = new URLSearchParams({
                    rru: 'addevent',
                    subject: tmtItem.name,
                    startdt: `${startDate}`,
                    enddt: `${endDate}`,
                    allday: true,
                    body: this.formatCalendarDescription(tmtItem),
                    location: tmtItem.location,
                });
                const url = `${baseUrl}?${queryParams.toString()}`;

                window.open(url, '_blank');

                this.closeAddToCalendarDialog();
            },
            openAddToCalendarDialog(tmtItem) {
                this.addToCalendarDialog = true;
                this.addToCalendarItem = tmtItem;
            },
            closeAddToCalendarDialog() {
                this.addToCalendarDialog = false;
                this.addToCalendarItem = null;
            },
            addToAppleCalendar(tmtItem) {
                this.addToIcal(tmtItem);
            },
            addToIcal(tmtItem) {
                const startDate = this.formatDateForGoogleCalendar(tmtItem.start_date);
                const endDate =  this.formatDateForGoogleCalendar(tmtItem.end_date, true);
                const url = `BEGIN:VCALENDAR
VERSION:2.0
BEGIN:VEVENT
SUMMARY:${tmtItem.name}
DESCRIPTION:${this.formatCalendarDescription(tmtItem)}
LOCATION:${tmtItem.location}
DTSTART:${startDate}
DTEND:${endDate}
END:VEVENT
END:VCALENDAR`;
                this.downloadIcsFile(url, 'bwf_calendar_event.ics');
                this.closeAddToCalendarDialog();
            },
            downloadIcsFile(icsData, fileName) {
                const blob = new Blob([icsData], { type: 'text/calendar' });
                const url = window.URL.createObjectURL(blob);
                const a = document.createElement('a');
                a.href = url;
                a.download = fileName;
                document.body.appendChild(a);
                a.click();
                document.body.removeChild(a);

            getWeekDates(weeksOffset = 0) {

                endOfWeek.setDate(startOfWeek.getDate() + 6);

                    startDate: this.formatDateYYYYMMDD(startOfWeek),
                    endDate: this.formatDateYYYYMMDD(endOfWeek),

            formatDateYYYYMMDD(date) {
                const year = date.getFullYear();
                const month = String(date.getMonth() + 1).padStart(2, '0'); // Months are zero-indexed
                const day = String(date.getDate()).padStart(2, '0');

                return `${year}-${month}-${day}`;

<script>
function loadLocalizedImages(countryCode) {
  var localizedImageElements = document.querySelectorAll('[data-localized-image]');
  var code = (countryCode || '').toLowerCase();
  var i;

  for (i = 0; i < localizedImageElements.length; i++) {
    var element = localizedImageElements[i];
    var srcFallback = element.getAttribute('data-src-fallback');
    var altFallback = element.getAttribute('data-alt-fallback');
    var src = element.getAttribute('data-src-' + code);
    var alt = element.getAttribute('data-alt-' + code);

    if (src || srcFallback) {
      element.src = src || srcFallback;
    }
    if (alt || altFallback) {
      element.alt = alt || altFallback;
    }

    element.setAttribute('data-localized-image', 'loaded');
  }
}

(function () {
  fetch('/wp-json/custom/v1/geolocation')
    .then(function (response) { return response.json(); })
    .then(function (data) { loadLocalizedImages(data.country_code); })
    .catch(function (error) {
      console.error('Error fetching geolocation:', error);
      loadLocalizedImages(null);
    });
})();

                 * unsettled read leaves the skeleton alone rather than asserting anything.
                 */
                function readFan() {
                    var api = window.urcAuth;
                    var state = api && typeof api.state === 'function' ? api.state() : null;

                    isAuth = !!(state && state.signedIn);
                    teamId = isAuth && state.clubId ? String(state.clubId) : '';
                    teamLogo = teamId && crests[teamId] ? crests[teamId] : '';

                 * sizing, padding and cyan tile are all inline. **The same declaration is in the

                var EMBLEM_STYLE = 'width: 27px; height: 27px; background-color: #00cde6; padding: 4px; border-radius: 8px 0;';

                var CREST_STYLE = 'width: 27px; height: 27px; padding: 4px; border-radius: 8px 0;';

                function writeEmblem(img) {

                 */
                function writeCrest(img) {
                    img.src = teamLogo;
                    img.alt = 'Team Logo';
                    img.className = 'auth0-team-logo';
                    img.setAttribute('data-team-id', teamId);
                    img.setAttribute('style', CREST_STYLE);

                        var logoImg = link.querySelector('img');

                            } else {

                function applyAccountRow() {

                            if (crest && teamId) {

                 * re-reads through `state()` instead: one source, so the two cannot disagree, and no
                 * dependence on an event shape a listener does not own.

                window.addEventListener('urc-auth:changed', apply);

                    window.urcAuth.subscribe(apply);
                }
